>>> from gestion.models import Cliente, Cuenta, Transaccion, Etiqueta
>>> cliente1 = Cliente.objects.create(nombre='Abel Lopez', email='abelopez@gmail.com', telefono='123456789')

>>> etiqueta1 = Etiqueta.objects.create(nombre='cuenta corriente')

>>> cuenta2.etiquetas.add(etiqueta2)
>>> Transaccion.objects.create(cuenta=cuenta1, tipo='DEPOSITO', monto=200000, descripcion='Deposito inicial')

RAW()
======
from gestion.models import Cliente
# Consulta que trae clientes que tienen teléfono registrado
consulta = "SELECT * FROM gestion_cliente WHERE telefono IS NOT NULL"
clientes_con_tel = Cliente.objects.raw(consulta)
for c in clientes_con_tel:
    print(c.id, c.nombre, c.telefono)

from django.db import connection
# Total de transacciones y monto acumulado
with connection.cursor() as cursor:
    cursor.execute("SELECT COUNT(*), COALESCE(SUM(monto), 0) FROM gestion_transaccion")
    conteo, acumulado = cursor.fetchone()
    print(f"Total movimientos: {conteo} | Acumulado: ${acumulado}")
